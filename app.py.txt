import google.generativeai as genai
import streamlit as st

# ضبط إعدادات الصفحة
st.set_page_config(
    page_title="المساعد الشيعي الجامع", page_icon="📖", layout="centered"
)

st.title("📖 المكتبة والمساعد الشيعي الذكي")
st.write(
    "إجابات واستدلالات معتمدة على المصادر الشيعية وآراء المراجع الأعلام."
)

# أدخل مفتاح الـ API الخاص بك هنا
API_KEY = "ضع_مفتاح_الـ_API_الخاص_بك_هنا"
genai.configure(api_key=API_KEY)

# تعليمات النظام لضبط الذكاء الاصطناعي
SYSTEM_INSTRUCTION = """
أنت مساعد ذكاء اصطناعي متخصص في العلوم والإلهيات والحديث والفقه الشيعي الإمامي.
وظيفتك:
1. الإجابة عن كافة الأسئلة والرد على الشبهات والمخالفين بالحجة والأدلة من القرآن الكريم والأحاديث المعتمدة.
2. الاعتماد الأساسي في الفتاوى والآراء على المراجع الأعلام: السيد علي السيستاني، السيد روح الله الخميني، والسيد علي الخامنئي.
3. الاستدلال بأمهات الكتب الشيعية (الكافي، الاستبصار، التهذيب، من لا حضره الفقيه، بحار الأنوار، وغيرها).
4. ذكر اسم الكتاب والأجزاء أو أرقام الأحاديث والمصادر بوضوح تام.
5. الالتزام بالحيادية العلمية والأدب الإسلامي الرفيع.
"""

# إعداد النموذج المجاني
model = genai.GenerativeModel(
    model_name="gemini-1.5-flash",
    system_instruction=SYSTEM_INSTRUCTION,
)

# حفظ سجل المحادثة
if "messages" not in st.session_state:
    st.session_state.messages = []

# عرض المحادثات السابقة
for message in st.session_state.messages:
    with st.chat_message(message["role"]):
        st.markdown(message["content"])

# استقبال سؤال المستخدم
if user_input := st.chat_input("اطرح سؤالك العقائدي أو الفقهي هنا..."):
    st.session_state.messages.append({"role": "user", "content": user_input})
    with st.chat_message("user"):
        st.markdown(user_input)

    # توليد الإجابة من Gemini
    with st.chat_message("assistant"):
        response = model.generate_content(user_input)
        st.markdown(response.text)
        st.session_state.messages.append(
            {"role": "assistant", "content": response.text}
        )
